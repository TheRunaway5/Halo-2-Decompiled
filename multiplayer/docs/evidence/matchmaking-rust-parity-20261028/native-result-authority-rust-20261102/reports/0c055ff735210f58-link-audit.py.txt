import hashlib,json,os,re,shlex,subprocess
from pathlib import Path

root=Path.cwd()
work=root/'build/test-results/native-result-authority-rust-20261102'
owner=root/'.local/test-scratch/native-result-authority-rust-20261102/halo2-test-t5cryxus'
source=owner/'native-source'
build=owner/'build'
scratch=Path(os.environ['HALO2_TEST_SCRATCH'])
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
read=lambda p:json.loads(p.read_text())
report={'passed':False,'scope':'Compiled native result bridge linkage and exact preserved modern C++ caller source; not original Live or delivery proof','scratch':str(scratch)}
try:
    before=work/'before-online-postgame.cpp.txt'
    assert sha(before)=='8e92b152036e3473f3cc3e6e2bdcd5182f0341285cfb72284d840e450cccacb0'
    reference=root/'tests/reference/native_results_before_rust_20261102/online_postgame_policy.cpp.inc'
    provenance=read(reference.parent/'provenance.json')
    assert sha(reference)==provenance['extraction_sha256']
    original=before.read_text()
    first=original.index('VistaPcrConsumedSnapshot online_postgame_publish(')
    tail=original.index('std::shared_ptr<const VistaPcrScreenOwner> online_postgame_screen(')
    assert original[first:tail]==reference.read_text()
    current=(root/'src/network/online_postgame.cpp').read_text()
    helper=original[original.index('VistaLoadingFloatingPoint floating_point()'):original.rfind('\n}\n',0,first)]
    assert helper in current
    assert original[tail:]==current[current.index('std::shared_ptr<const VistaPcrScreenOwner> online_postgame_screen('):]
    assert sha(source/'include/halo2/online_postgame.hpp')==sha(root/'include/halo2/online_postgame.hpp')
    entry=next(e for e in read(build/'compile_commands.json') if e['file'].endswith('/src/network/online_postgame.cpp'))
    command=shlex.split(entry['command'])
    original_object=build/entry['output']
    assert original_object.is_file()
    preimage=scratch/'before.cpp';preimage.write_bytes(before.read_bytes())
    command[command.index('-o')+1]=str(scratch/'before.o')
    command[command.index('-c')+1]=str(preimage)
    run=subprocess.run(command,cwd=entry['directory'],capture_output=True)
    log=run.stdout+run.stderr
    (work/'link-audit-compiler.log').write_bytes(log[-32768:])
    report['reference_compile']={'command':command,'exit_code':run.returncode,'log_bytes':len(log),'log_sha256':hashlib.sha256(log).hexdigest()}
    assert run.returncode==0
    symbols={}
    for label,path in [('cpp_before',scratch/'before.o'),('rust_wired',original_object)]:
        nm=subprocess.run(['nm','-u','-C',str(path)],capture_output=True,text=True,check=True).stdout
        symbols[label]={'object_sha256':sha(path),'object_bytes':path.stat().st_size,
            'rust_result_calls':[line.strip() for line in nm.splitlines() if '::native_result_' in line],
            'all_undefined_symbols_sha256':hashlib.sha256(nm.encode()).hexdigest()}
    assert not symbols['cpp_before']['rust_result_calls']
    calls=symbols['rust_wired']['rust_result_calls']
    assert len(calls)==2 and all(any(name in line for line in calls) for name in ['native_result_publish(', 'native_result_validate('])
    rust=read(work/'full-rust/report.json')['source_pins']
    native={}
    for p in sorted((work/'native').glob('iteration-2-source-pins-*.json')):native.update(read(p))
    common=sorted(set(rust)&set(native))
    mismatch=[n for n in common if rust[n]!=native[n]]
    assert not mismatch,mismatch
    missing_native=sorted(set(rust)-set(native))
    late=[]
    for n in missing_native:
        assert not n.startswith(('rust/','multiplayer/master_server/')),n
        row={'path':n,'rust_sha256':rust[n],'current_sha256':sha(root/n),'borrowed_reference':(source/n).is_symlink()}
        assert row['borrowed_reference'] or (n=='tests/fixtures/weapon_effect_location_copy_tu15_original.json.gz' and not (source/n).exists()),n
        if row['current_sha256']!=rust[n]:
            assert n=='tests/fixtures/weapon_effect_location_copy_tu15_original.json.gz',n
            needles=[Path(n).name,'weapon_effect_location_copy_tu15']
            references=[]
            for path in (root/'rust').rglob('*'):
                if path.is_file() and path.suffix in ('.rs','.toml','.cpp','.hpp') and 'target' not in path.parts:
                    if any(needle in path.read_text(errors='replace') for needle in needles):references.append(str(path.relative_to(root)))
            assert not references,references
            row.update(unrelated_foreign_reference_drift=True,rust_source_references=references,
                excluded_from_current_qualification=True,reason='Unrelated weapon fixture added after native freeze and changed concurrently; no Rust crate or result fixture reference')
        late.append(row)
    critical=['src/network/online_postgame.cpp','include/halo2/online_postgame.hpp',
        'rust/online_protocol/src/matchmaking_native_results.rs','rust/online_protocol/tests/matchmaking_native_results.rs',
        'rust/online_protocol/src/lib.rs','rust/online_protocol_bridge/src/matchmaking_native_results.rs',
        'rust/online_protocol_bridge/src/matchmaking_native_results.hpp','rust/online_protocol_bridge/src/lib.rs',
        'rust/online_protocol_bridge/build.rs','cmake/OnlineProtocolRust.cmake','cmake/OnlinePeers.cmake',
        'tools/build_online_protocol_rust.py','tests/test_build_online_protocol_rust.py',
        'tests/native_result_authority_tests.cpp','tests/test_native_result_authority.py',
        'tests/reference/native_results_before_rust_20261102/online_postgame_policy.cpp.inc',
        'tests/reference/native_results_before_rust_20261102/provenance.json']
    pins={n:sha(root/n) for n in critical}
    assert all(native.get(n)==h for n,h in pins.items())
    assert 'unsafe' not in re.sub(r'//[^\n]*','',(root/'rust/online_protocol/src/matchmaking_native_results.rs').read_text())
    counts={}
    for name in ['online_protocol','online_protocol_bridge','matchmaking_service']:
        text=(work/f'full-rust/{name}-test.log').read_text()
        rows=re.findall(r'test result: ok\. (\d+) passed; 0 failed; (\d+) ignored;',text)
        counts[name]={'passed':sum(int(n) for n,_ in rows),'ignored':sum(int(n) for _,n in rows)}
    assert [counts[n]['passed'] for n in counts]==[493,65,20],counts
    report.update(passed=True,source_pins=pins,symbols=symbols,
        exact_reference_functions_preserved=True,original_ranking_floating_point_helpers_and_presentation_tail_unchanged=True,
        public_header_unchanged=True,rust_native_shared_inputs=len(common),rust_native_shared_input_mismatches=mismatch,
        rust_inputs_not_copied_to_native=late,rust_counts=counts,
        cpp_before_sha256=sha(before),reference=provenance,native_compile_command=entry,
        audit_harness_corrections=['Restrict unchanged helper extraction to its unnamed namespace boundary',
            'Separate borrowed binary references from copied native source inputs and exclude one unused concurrent weapon fixture'])
finally:
    (work/'link-audit.json').write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps({k:report[k] for k in ['passed','rust_native_shared_inputs','rust_counts']}))

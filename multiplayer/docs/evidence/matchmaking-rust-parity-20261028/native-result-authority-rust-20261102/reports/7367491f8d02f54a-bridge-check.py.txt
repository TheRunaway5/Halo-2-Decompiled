import os,shutil,subprocess,json,hashlib,sys
from pathlib import Path
root=Path.cwd();scratch=Path(os.environ["HALO2_TEST_SCRATCH"]);source=scratch/"source";source.mkdir()
for name in ("online_protocol","online_protocol_bridge"):shutil.copytree(root/"rust"/name,source/"rust"/name,ignore=shutil.ignore_patterns("target","__pycache__"))
pins={str(p.relative_to(source)):hashlib.sha256(p.read_bytes()).hexdigest() for p in source.rglob("*") if p.is_file()}
env=dict(os.environ,CARGO_TARGET_DIR=str(scratch/"cargo"),CARGO_BUILD_JOBS="2",RUSTUP_AUTO_INSTALL="0",RUSTUP_TOOLCHAIN="1.96.0")
command=["cargo","check","--locked","--offline","--manifest-path",str(source/"rust/online_protocol_bridge/Cargo.toml")]
r=subprocess.run(command,env=env,stdout=subprocess.PIPE,stderr=subprocess.STDOUT)
out=root/"build/test-results/native-result-authority-rust-20261102";(out/"bridge-check.log").write_bytes(r.stdout[-1048576:]);(out/"bridge-check.json").write_text(json.dumps({"passed":r.returncode==0,"command":command,"exit_code":r.returncode,"source_pins":pins,"scratch":str(scratch)},indent=2)+"\n")
print(r.stdout.decode(errors="replace")[-8000:],flush=True);sys.exit(r.returncode)

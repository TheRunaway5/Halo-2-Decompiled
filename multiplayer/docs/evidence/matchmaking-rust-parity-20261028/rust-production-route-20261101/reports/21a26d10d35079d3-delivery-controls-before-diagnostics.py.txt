def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def port():
    with socket.socket() as listener:
        listener.bind(("127.0.0.1", 0))
        return listener.getsockname()[1]


class Process:
    def __init__(self, command, log, *, env, cwd):
        self.log = log
        self.stream = log.open("ab")
        self.process = subprocess.Popen(command, stdout=self.stream, stderr=subprocess.STDOUT, env=env, cwd=cwd)

    def close(self):
        if self.process.poll() is None:
            self.process.terminate()
            try:
                self.process.wait(timeout=8)
            except subprocess.TimeoutExpired:
                self.process.kill()
                self.process.wait()
        self.stream.close()


def tls_files(scratch):
    from cryptography import x509
    from cryptography.hazmat.primitives import hashes, serialization
    from cryptography.hazmat.primitives.asymmetric import rsa
    now = datetime.datetime.now(datetime.timezone.utc)
    key = rsa.generate_private_key(public_exponent=65537, key_size=2048)
    name = x509.Name([x509.NameAttribute(x509.NameOID.COMMON_NAME, "Private Halo2 gateway fixture")])
    certificate = (x509.CertificateBuilder().subject_name(name).issuer_name(name).public_key(key.public_key())
                   .serial_number(x509.random_serial_number()).not_valid_before(now - datetime.timedelta(minutes=1))
                   .not_valid_after(now + datetime.timedelta(hours=4))
                   .add_extension(x509.BasicConstraints(ca=True, path_length=0), critical=True)
                   .add_extension(x509.SubjectAlternativeName([x509.DNSName("localhost"),
                                                              x509.IPAddress(ipaddress.ip_address("127.0.0.1"))]), critical=False)
                   .sign(key, hashes.SHA256()))
    cert, private = scratch / "test-ca.pem", scratch / "test-key.pem"
    cert.write_bytes(certificate.public_bytes(serialization.Encoding.PEM))
    private.write_bytes(key.private_bytes(serialization.Encoding.PEM, serialization.PrivateFormat.PKCS8,
                                          serialization.NoEncryption()))
    private.chmod(0o600)
    return cert, private


class Gateway:
    def __init__(self, binary, source, scratch, backend, directory_port, endpoint, cert, key, *, old_limits=False):
        self.port = endpoint
        self.context = ssl.create_default_context(cafile=str(cert))
        config = (source / "multiplayer/master_server/Caddyfile.rust").read_text()
        assert config.count("admin off") == 1 and config.count("{$MASTER_DOMAIN} {") == 1
        # The fixture adds only local certificate/listen settings. Routing,
        # body limits and upstream transport remain the production file.
        config = config.replace("admin off", "admin off\n\tauto_https off")
        config = config.replace("{$MASTER_DOMAIN} {", "{$MASTER_DOMAIN} {\n\tbind 127.0.0.1\n\ttls " + str(cert) + " " + str(key))
        if old_limits:
            config = config.replace("response_header_timeout 100s", "response_header_timeout 10s").replace("max_size 49152", "max_size 16384")
        path = scratch / ("gateway-before.caddy" if old_limits else "gateway.caddy")
        path.write_text(config)
        env = dict(os.environ, MASTER_DOMAIN=f"https://localhost:{endpoint}",
                   H2_MATCHMAKING_UPSTREAM=f"127.0.0.1:{backend.port}", H2_DIRECTORY_UPSTREAM=f"127.0.0.1:{directory_port}",
                   XDG_DATA_HOME=str(scratch / "caddy-data"), XDG_CONFIG_HOME=str(scratch / "caddy-config"))
        adapted = subprocess.run([str(binary), "adapt", "--config", str(path), "--adapter", "caddyfile"],
                                 env=env, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
        assert adapted.returncode == 0, adapted.stderr.decode()[-2000:]
        self.adapted = json.loads(adapted.stdout)
        self.owner = Process([str(binary), "run", "--config", str(path), "--adapter", "caddyfile"],
                             scratch / "gateway.log", env=env, cwd=source)
        try:
            deadline = time.monotonic() + 15
            while True:
                try:
                    if self.request("GET", "/v1/health")[0] == 200:
                        break
                except (OSError, http.client.HTTPException):
                    pass
                assert self.owner.process.poll() is None, self.owner.log.read_text()[-2000:]
                assert time.monotonic() < deadline, "Gateway did not become available"
                time.sleep(0.05)
        except BaseException:
            self.close()
            raise

    def request(self, method, path, *, body=None, token=None, raw=None, timeout=5):
        connection = http.client.HTTPSConnection("localhost", self.port, context=self.context, timeout=timeout)
        headers = {"Content-Type": "application/json"}
        if token:
            headers["Authorization"] = "Bearer " + token
        data = raw if raw is not None else json.dumps(body).encode() if body is not None else None
        try:
            connection.request(method, path, data, headers)
            response = connection.getresponse()
            data = response.read(2 * 1024 * 1024 + 1)
            assert len(data) <= 2 * 1024 * 1024
            try:
                value = json.loads(data)
            except ValueError:
                value = {"proxy_error": data.decode(errors="replace")[:200]}
            return response.status, value
        finally:
            connection.close()

    def close(self):
        self.owner.close()


def qualify(source, scratch, binary, library, caddy, shim, output):

    state = scratch / "state"

    env = {k: v for k, v in os.environ.items() if not k.startswith("H2_") and k not in ("STEAM_KEY", "LD_PRELOAD")}
    env.update(H2_MASTER_STATE=str(state), H2_MASTER_BIND="127.0.0.1", H2_MASTER_PORT=str(directory_port),
               H2_MASTER_PUBLIC_URL=issuer, H2_MASTER_BEHIND_TLS_PROXY="1", H2_DIRECTORY_ONLY="1",
               H2_SECURITY_LIBRARY=str(library), HALO2_MATCHMAKING_TEST_INPUTS="0")
    rust_env = dict(LD_PRELOAD=str(shim), H2_STEAM_KEY="1" * 32, H2_TEST_STEAM_PROVIDER_DATA=str(provider),
                    H2_TEST_STEAM_PROVIDER_CALLS=str(calls), H2_PUBLIC_URL=issuer, H2_EOS_OPENID_AUDIENCE="private-delivery-fixture")
    directory = Process([sys.executable, "-B", "-m", "multiplayer.master_server"], scratch / "directory.log", env=env, cwd=source)
    backend = gateway = None
    tokens = ["h2s_" + letter * 43 for letter in "PQ"]
    subjects = ["76561197960287930", "76561197960287931"]

    def sql(statement, params=()):
        with sqlite3.connect(state / "registry.sqlite3") as db:
            return db.execute(statement, params).fetchall()

    def expect(method, path, status=200, **kw):
        actual, value = gateway.request(method, path, **kw)
        assert actual == status, (method, path, actual, status, value.get("error", value.get("proxy_error")))
        return value

    def start(old_limits=False):
        nonlocal backend, gateway
        backend = service.Server(binary, state, scratch, configured, environment=rust_env, tls=True)
        gateway = Gateway(caddy, source, scratch, backend, directory_port, endpoint, cert, key, old_limits=old_limits)
        if not old_limits:
            (output / "caddy-adapted.json").write_text(json.dumps(gateway.adapted, indent=2) + "\n")

    def stop():
        nonlocal backend, gateway
        if gateway:
            gateway.close()
            gateway = None
        if backend:
            backend.close()
            backend = None

    def login(index=0):
        return expect("POST", "/v1/account/session", 201, token=tokens[index],
                      body={"screen_name": "Delivery Pilot " + str(index)})["token"]

    def queue(index=0, compatibility="v16"):
        entries = json.loads(configured.read_text())["playlists"]
        maps = sorted({game["map"] for entry in entries for game in entry["games"]})
        return expect("POST", "/v1/matchmaking/session", 201, token=tokens[index], body=dict(
            screen_name="Delivery Pilot " + str(index), maps=maps, compatibility="halo2-native-queue-" + compatibility,
            language=0))["token"]

        return expect("GET", "/v1/party?privacy=1", token=grant)["party"]

        if action not in ("create", "accept") and "version" not in values:

        return expect("POST", "/v1/party/commands", status, token=grant,

    try:

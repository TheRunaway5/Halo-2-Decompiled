def sections(path):
 text=path.read_text();out={}
 # CTest writes Start and result records in sequential --parallel 1 order.
 rows=list(re.finditer(r'^\s*Start\s+\d+:\s+(\w+)\s*$',text,re.M))
 for index,row in enumerate(rows):
  part=text[row.end():rows[index+1].start() if index+1<len(rows) else len(text)]
  result=re.search(r'Test\s+#\d+:\s+\w+\s+\.+\s*(Passed|\*\*\*Failed|\*\*\*Not Run|\*\*\*Skipped)',part)
  assert result,(row.group(1),part[:300])
  out[row.group(1)]={'passed':result.group(1)=='Passed','skipped':result.group(1) not in ('Passed','***Failed'),'text':part}
 return out

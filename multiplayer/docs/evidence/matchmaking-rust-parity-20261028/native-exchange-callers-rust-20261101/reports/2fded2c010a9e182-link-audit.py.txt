critical=['src/network/online_match_session.cpp','rust/online_protocol/src/matchmaking_native_exchange.rs',

    'rust/online_protocol/tests/matchmaking_native_session.rs',
    'rust/online_protocol_bridge/src/matchmaking_native_exchange.rs','rust/online_protocol_bridge/src/matchmaking_native_exchange.hpp',
    'rust/online_protocol_bridge/src/matchmaking_native_session.rs','rust/online_protocol_bridge/src/lib.rs',
    'rust/online_protocol_bridge/build.rs','cmake/OnlineProtocolRust.cmake','tools/build_online_protocol_rust.py',

assert native['baseline']==expected_baseline

    'baseline_cpp_caller':expected_baseline,'command':command,'object_sha256':sha(objects[0]),'rust_entries':entries,
    'cpp_parser_crypto_dependencies':parser_crypto,'rust_native_source_count':len(rust['source_pins']),
    'final_rust_native_source_mismatch':rust_mismatch,'current_critical_source_drift':critical_drift,
    'source_pins':{n:pins[n] for n in critical},'audit_scratch':os.environ['HALO2_TEST_SCRATCH'],

    'retained_large_artifacts':[]}
result['passed']=result['passed'] and not parser_crypto and not rust_mismatch and (expected_baseline or not critical_drift)
(work/('baseline-link-audit.json' if expected_baseline else 'link-audit.json')).write_text(json.dumps(result,indent=2)+'\n')
if not expected_baseline:(work/'current-source-drift.json').write_text(json.dumps(drift,indent=2)+'\n')
print(json.dumps({'passed':result['passed'],'baseline':expected_baseline,'entries':entries,'rust_mismatch':rust_mismatch,'critical_drift':critical_drift,'drift_count':len(drift)}));assert result['passed']

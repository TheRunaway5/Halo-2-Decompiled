import hashlib,json,os,re,stat
from pathlib import Path
root=Path.cwd();work=root/'build/test-results/frontend-couch-flow-rust-20261102';read=lambda n:json.loads((work/n).read_text());iteration=read('accepted-native-iteration.json')['iteration'];record=read(f'native/iteration-{iteration}-report.json');assert record['passed']and record['frozen_inputs_intact']and not(work/f'native/iteration-{iteration}-running.json').exists()
log=(work/f'native/iteration-{iteration}/tests.log').read_text();passed=re.findall(r'Test\s+#\d+:\s+(\S+)\s+\.+\s+Passed\s+([\d.]+) sec',log);expected=read('focused-native-expected-tests.json');assert len(passed)==len(expected)==13 and {n for n,_ in passed}==set(expected);assert not any(x in log for x in['Skipped','Not Run','***Failed'])
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest();native={}
for n in record['source_pin_parts']:native.update(read('native/'+n))
names=read('declared-scope.json');assert all(sha(root/n)==native.get(n)for n in names);scratch=Path(record['scratch']);build=scratch/'build';objects=read('production-objects.json');assert objects['passed']and not Path(objects['scratch']).exists();archive=objects['cmake_abi_archive'];assert sha(build/archive['path'])==archive['sha256']
for n,receipt in record['binaries'].items():assert sha(build/n)==receipt['sha256']
size=count=0
for current,dirs,files in os.walk(scratch,followlinks=False):
 dirs[:]=[d for d in dirs if not(Path(current)/d).is_symlink()]
 for n in files:
  s=(Path(current)/n).lstat()
  if stat.S_ISREG(s.st_mode):size+=s.st_size;count+=1
report=dict(scratch=str(scratch),regular_bytes=size,regular_files=count,borrowed_symlinks_not_followed=True,accepted_iteration=iteration,passing_focused_cmake_checks=13,whole_native_runtime_checklist_completed=False,scoped_sources_current=True,scoped_sources=len(names),validated_object_hashes_unchanged=True,actual_cmake_abi_archive=archive,production_probe_objects_removed=True,retained_large_artifacts=[]);(work/'native-before-cleanup.json').write_text(json.dumps(report,indent=2)+'\n')
with(scratch/'control.fifo').open('w')as f:f.write(json.dumps(dict(action='quit'))+'\n')
print(json.dumps(dict(passing_focused_cmake_checks=13,regular_bytes_pending_cleanup=size,quit_sent=True)))

async def qualify(scenario):
    base=f"http://{server.host}:{server.port}"
    async with aiohttp.ClientSession() as http_client:
        async def http(method,path,status,grant=None,body=None):
            headers={"Authorization":"Bearer "+grant} if grant else {}
            async with http_client.request(method,base+path,headers=headers,json=body) as reply:
                data=await reply.json()
                assert reply.status==status,(method,path,reply.status,data.get("error"))
                return data
        auth=[(await http("POST","/v1/account/session",201,t,{"screen_name":name}))["token"] for t,name in zip(tokens,names)]
        with sqlite3.connect(db) as connection:
            connection.executemany("UPDATE player_profiles SET discord_username=? WHERE player=(SELECT player FROM player_identities WHERE provider='steam' AND subject=?)",
                [(f"party_review_{i}",s) for i,s in enumerate(subjects)])
        grants=[(await http("POST","/v1/matchmaking/session",201,token,dict(screen_name=name,maps=["lockout"],
            compatibility="halo2-native-queue-v12",language=0)))["token"] for token,name in zip(tokens,names)]
        chosen=dict(playlist_id=entry["id"],revision=entry["revision"])
        async def party(i):
            return await http("GET","/v1/party",200,auth[i])
        async def command(i,action,**fields):
            if action not in ("create","accept"):
                fields["version"]=(await party(i))["party"]["version"]
            return await http("POST","/v1/party/commands",200,auth[i],
                dict(action=action,request_id=str(uuid.uuid4()))|fields)
        for leader,follower in ((0,1),(2,3)):
            await command(leader,"create")
            await command(leader,"invite",username=f"party_review_{follower}")
            invite=(await party(follower))["invitations"][0]
            await command(follower,"accept",target=invite["id"],version=invite["version"])
            await http("POST","/v1/matchmaking/ticket",409,grants[leader],chosen|dict(request_id=str(uuid.uuid4())))
            await command(follower,"ready",**chosen)
            await http("POST","/v1/matchmaking/ticket",201,grants[leader],chosen|dict(request_id=str(uuid.uuid4())))
        descriptors=[await http("GET","/v1/matchmaking/transport",200,g) for g in grants]
        room=descriptors[0]["id"]
        assert all(d["id"]==room and d["transport"]["kind"]=="direct-udp-v1" for d in descriptors)
        roster=descriptors[0]["players"]
        assert len(roster)==4 and len({p["identity"] for p in roster})==4
        groups=[tuple(roster[d["slot"]]["team"] for d in descriptors[i:i+2]) for i in (0,2)]
        assert groups[0][0]==groups[0][1] and groups[1][0]==groups[1][1] and groups[0][0]!=groups[1][0]
        
        queues=[asyncio.Queue() for _ in grants]
        sockets=[]
        readers=[]
        udp=[socket.socket(socket.AF_INET,socket.SOCK_DGRAM) for _ in grants]
        for wire in udp:
            wire.bind(("127.0.0.1",0))
            wire.setblocking(False)
        async def read(ws,queue):
            async for message in ws:
                if message.type==aiohttp.WSMsgType.TEXT:
                    item=json.loads(message.data)
                    if "ping" in item:
                        await ws.send_json({"pong":item["ping"]})
                    else:
                        await queue.put(item)
            await queue.put(None)
        async def until(queue,field,value=None,timeout=8):
            async with asyncio.timeout(timeout):
                while True:
                    item=await queue.get()
                    assert item is not None,("Room closed",field,value)
                    if field in item and (value is None or item[field]==value):
                        return item
        try:
            for descriptor,queue in zip(descriptors,queues):
                ws=await http_client.ws_connect(base+f"/v1/matches/{room}/signal?completion=1&native_handoff=1",
                    headers={"Authorization":"Bearer "+descriptor["token"]},compress=0)
                sockets.append(ws)
                readers.append(asyncio.create_task(read(ws,queue)))
            rtts=[]
            loop=asyncio.get_running_loop()
            for i,wire in enumerate(udp):
                rows=[]
                for j,peer in enumerate(udp):
                    if i==j:
                        continue
                    challenge=os.urandom(12)
                    start=time.monotonic_ns()
                    await loop.sock_sendto(wire,challenge,peer.getsockname())
                    sample,address=await asyncio.wait_for(loop.sock_recvfrom(peer,32),2)
                    assert sample==challenge and address==wire.getsockname()
                    await loop.sock_sendto(peer,sample,address)
                    assert await asyncio.wait_for(loop.sock_recvfrom(wire,32),2)==(challenge,peer.getsockname())
                    rows.append(dict(peer=descriptors[j]["slot"],rtt_ms=math.ceil((time.monotonic_ns()-start)/1_000_000)))
                rtts.append(rows)
            for ws,rows in zip(sockets,rtts):
                await ws.send_json({"peer_ready":rows})
            loading=await asyncio.gather(*(until(queue,"phase","loading") for queue in queues))
            host=loading[0]["host"]
            assert all(row["host"]==host for row in loading)
            for grant in grants:
                row=await http("GET","/v1/matchmaking/ticket?start=1&start_room=1",200,grant)
                assert row["start_room"]==room and 0<row["start_countdown_ms"]<=10000
            for ws in sockets:
                await ws.send_json({"loaded":"0123456789abcdef:000001a"})
            await asyncio.gather(*(until(queue,"phase","running",14) for queue in queues))
            
            publisher=next(i for i,d in enumerate(descriptors) if d["slot"]==host)
            raw=bytearray(0xdc78)
            raw[:2]=raw[0x10:0x12]=b"\x01\x01"
            raw[0x64]=2
            raw[0x13]=raw[0x68]=1
            teams=sorted({p["team"] for p in roster})
            for placement,team in enumerate(teams):
                at=0xcb0+team*0x18
                raw[at]=1
                raw[at+1]=placement
                struct.pack_into("<h",raw,at+2,10-placement)
            for slot,player in enumerate(roster):
                at=0x370+slot*0x94
                raw[at]=1
                raw[at+2:at+10]=bytes.fromhex(player["identity"])
                name=player["name"].encode("utf-16-le")
                raw[at+12:at+12+len(name)]=name
                raw[at+0x88]=player["team"]
                raw[at+0x90]=slot
                struct.pack_into("<h",raw,at+0x92,4+(slot%2))
            digest=hashlib.sha256(raw).hexdigest()
            await sockets[publisher].send_json({"postgame":raw.hex()})
            reviews=await asyncio.gather(*(until(q,"postgame") for q in queues))
            assert all(row["postgame"]==raw.hex() for row in reviews)
            disputing=(publisher+1)%len(sockets)
            survivors=[i for i in range(len(sockets)) if i!=disputing]
            # Cross-socket delivery provides a real fence for each earlier vote.
            for i in survivors:
                await sockets[i].send_json({"confirm_result":digest})
                nonce=f"review-fence-{i}"
                await sockets[i].send_json({"peer_signal":dict(peer=descriptors[disputing]["slot"],
                    type="candidate",payload=nonce,mid="0")})
                fence=(await until(queues[disputing],"peer_signal"))["peer_signal"]
                assert fence["payload"]==nonce and fence["peer"]==descriptors[i]["slot"]
            assert sql("SELECT count(*) FROM match_results")==[(0,)]
            await sockets[disputing].send_json({"confirm_result":hashlib.sha256(b"conflicting terminal observation").hexdigest()})
            await asyncio.wait_for(readers[disputing],5)
            assert sockets[disputing].closed
            recipients=await asyncio.gather(*(until(queues[i],"completed",digest) for i in survivors))
            assert recipients==[dict(completed=digest,recorded=False)]*3

            returning=survivors[-1]
            await sockets[returning].close()
            await readers[returning]
            replacement=await http_client.ws_connect(
                base+f"/v1/matches/{room}/signal?completion=1&native_handoff=1",
                headers={"Authorization":"Bearer "+descriptors[returning]["token"]},compress=0)
            sockets[returning]=replacement
            queues[returning]=asyncio.Queue()
            readers[returning]=asyncio.create_task(read(replacement,queues[returning]))
            assert (await until(queues[returning],"postgame"))["postgame"]==raw.hex()
            await replacement.send_json({"confirm_result":digest})
            observed=None
            try:
                observed=await until(queues[returning],"completed",digest)
            except AssertionError as error:
                assert error.args==(("Room closed","completed",digest),),error
                assert replacement.closed
            counts={name:sql("SELECT count(*) FROM "+name)[0][0] for name in
                ("match_results","match_result_players","match_result_memberships","match_result_skill","carnage_reports")}
            assert set(counts.values())=={0}
            assert sql("SELECT xp FROM playlist_skill ORDER BY xp")==[(0,)]*4
            return dict(survivor_completions=len(recipients),matched_postgame_on_reconnect=True,
                retry_disconnected=replacement.closed,recovered_noncompetitive_completion=observed==dict(completed=digest,recorded=False),
                persisted=counts,loopback_probe_exchanges=sum(map(len,rtts)))
        finally:
            for wire in udp:
                wire.close()
            for ws in sockets:
                await ws.close()
            await asyncio.gather(*readers)


def seed():
    with sqlite3.connect(db) as connection:
        connection.executemany("INSERT INTO steam_login_sessions VALUES(?,?,?,?)",[
            (hashlib.sha256(token.encode()).hexdigest(),subject,"Private party fixture",int(time.time())+3600)
            for token,subject in zip(tokens,subjects)])


async def run_legacy(scenario):
    from aiohttp.test_utils import TestServer
    from multiplayer.master_server.service import Settings, create_app
    global server
    settings=Settings(state=state,public_url="http://127.0.0.1:8080",allow_http=True,steam_enabled=True,

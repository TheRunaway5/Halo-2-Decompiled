cmake_minimum_required(VERSION 3.24)
get_filename_component(HALO2_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${HALO2_ROOT}/cmake/macos/Defaults.cmake")
project(Halo2MacOSIceTests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(nlohmann_json REQUIRED CONFIG)
include("${HALO2_ROOT}/cmake/TestStorage.cmake")
include(CTest)
include(FetchContent)
set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)
set(BUILD_SHARED_LIBS OFF)
set(USE_NETTLE OFF)
set(NO_TESTS ON)
FetchContent_Declare(halo2_ice
    GIT_REPOSITORY https://github.com/paullouisageneau/libjuice.git
    GIT_TAG b89c792e3612faf2f12cf35bcc56857313a06be3
    GIT_SHALLOW FALSE EXCLUDE_FROM_ALL)
FetchContent_MakeAvailable(halo2_ice)
execute_process(COMMAND git -C "${halo2_ice_SOURCE_DIR}" rev-parse HEAD OUTPUT_VARIABLE revision OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT revision STREQUAL "b89c792e3612faf2f12cf35bcc56857313a06be3")
    message(FATAL_ERROR "The ICE diagnostic source differs from the engine's pinned submodule")
endif()
set(source_hashes)
foreach(path src/agent.c src/conn_poll.c src/udp.c src/stun.c src/addr.c src/thread.h src/timestamp.c include/juice/juice.h)
    file(SHA256 "${halo2_ice_SOURCE_DIR}/${path}" hash)
    list(APPEND source_hashes "\"${path}\":\"${hash}\"")
endforeach()
list(JOIN source_hashes "," source_hashes)
file(WRITE "${CMAKE_BINARY_DIR}/ice-source.json" "{\"revision\":\"${revision}\",\"files\":{${source_hashes}}}\n")
add_executable(macos_ice_tests ice_tests.cpp)
target_link_libraries(macos_ice_tests PRIVATE LibJuice::LibJuiceStatic nlohmann_json::nlohmann_json)
if(APPLE)
    foreach(arch IN LISTS CMAKE_OSX_ARCHITECTURES)
        add_test(NAME macos_ice_poll_${arch} COMMAND /usr/bin/arch -${arch} $<TARGET_FILE:macos_ice_tests>
            "${CMAKE_BINARY_DIR}/ice-${arch}.json")
    endforeach()
else()
    add_test(NAME macos_ice_poll_linux COMMAND macos_ice_tests "${CMAKE_BINARY_DIR}/ice-linux.json")
endif()

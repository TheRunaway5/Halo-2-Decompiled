cmake_minimum_required(VERSION 3.24)
get_filename_component(HALO2_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${HALO2_ROOT}/cmake/macos/Defaults.cmake")
project(Halo2SoundStorageTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
include("${HALO2_ROOT}/cmake/TestStorage.cmake")
include(CTest)
add_library(sound_storage STATIC "${HALO2_ROOT}/src/audio/native_sound_storage.cpp" "${HALO2_ROOT}/src/audio/sound_cache_startup.cpp")
target_include_directories(sound_storage PUBLIC "${HALO2_ROOT}/include" "${HALO2_ROOT}/src")
option(HALO2_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
if(HALO2_SANITIZERS)
    target_compile_options(sound_storage PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(sound_storage PUBLIC -fsanitize=address,undefined)
endif()
add_executable(sound_storage_granularity_tests granularity_tests.cpp)
target_link_libraries(sound_storage_granularity_tests PRIVATE sound_storage)
add_executable(native_sound_storage_tests "${HALO2_ROOT}/tests/native_sound_storage_tests.cpp")
target_link_libraries(native_sound_storage_tests PRIVATE sound_storage)
set(HALO2_SOUND_CACHE_REFERENCE "${HALO2_ROOT}/analysis/multiplayer/integration/sound-cache-startup-native/fixtures.tsv"
    CACHE FILEPATH "Authenticated original sound cache constructor fixture")
if(NOT EXISTS "${HALO2_SOUND_CACHE_REFERENCE}")
    message(FATAL_ERROR "Supply the authenticated original constructor fixture with HALO2_SOUND_CACHE_REFERENCE")
endif()
if(APPLE)
    foreach(target sound_storage_granularity_tests native_sound_storage_tests)
        add_custom_command(TARGET ${target} POST_BUILD COMMAND /usr/bin/lipo -archs $<TARGET_FILE:${target}>)
    endforeach()
    foreach(arch IN LISTS CMAKE_OSX_ARCHITECTURES)
        add_test(NAME sound_storage_original_${arch} COMMAND /usr/bin/arch -${arch}
            $<TARGET_FILE:native_sound_storage_tests> "${HALO2_SOUND_CACHE_REFERENCE}"
            "${CMAKE_BINARY_DIR}/sound-storage-original-${arch}.json")
        add_test(NAME sound_storage_${arch} COMMAND /usr/bin/arch -${arch}
            $<TARGET_FILE:sound_storage_granularity_tests> "${CMAKE_BINARY_DIR}/sound-storage-${arch}.json")
        foreach(alias 82e44000 82e45000)
            add_test(NAME sound_storage_${alias}_${arch} COMMAND /usr/bin/arch -${arch}
                $<TARGET_FILE:sound_storage_granularity_tests> "${CMAKE_BINARY_DIR}/sound-storage-${alias}-${arch}.json" --map-only ${alias})
        endforeach()
    endforeach()
else()
    add_test(NAME sound_storage_original_host COMMAND native_sound_storage_tests "${HALO2_SOUND_CACHE_REFERENCE}"
        "${CMAKE_BINARY_DIR}/sound-storage-original-host.json")
    add_test(NAME sound_storage_host COMMAND sound_storage_granularity_tests "${CMAKE_BINARY_DIR}/sound-storage-host.json")
    foreach(alias 82e44000 82e45000)
        add_test(NAME sound_storage_${alias}_host COMMAND sound_storage_granularity_tests "${CMAKE_BINARY_DIR}/sound-storage-${alias}-host.json" --map-only ${alias})
    endforeach()
endif()

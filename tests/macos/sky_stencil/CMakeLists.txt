cmake_minimum_required(VERSION 3.24)
get_filename_component(HALO2_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${HALO2_ROOT}/cmake/macos/Defaults.cmake")
project(Halo2MacOSSkyStencilTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(nlohmann_json REQUIRED CONFIG)
include("${HALO2_ROOT}/cmake/Glm.cmake")
include("${HALO2_ROOT}/cmake/TestStorage.cmake")
include(CTest)
if(APPLE)
    include("${HALO2_ROOT}/cmake/macos/Platform.cmake")
else()
    find_package(Vulkan REQUIRED)
endif()
find_program(GLSLANG_VALIDATOR glslangValidator REQUIRED)
set(HALO2_SKY_REFERENCE_ROOT "" CACHE PATH "Owned authenticated original sky reference root")
if(NOT HALO2_SKY_REFERENCE_ROOT)
    message(FATAL_ERROR "Supply the authenticated reference tree prepared in owned scratch")
endif()
include("${HALO2_ROOT}/cmake/SourceSkyStencilReference.cmake")
halo2_verify_source_sky_references("${HALO2_SKY_REFERENCE_ROOT}")
include("${HALO2_ROOT}/cmake/SourceSkyStencil.cmake")
halo2_add_source_sky_stencil("${HALO2_ROOT}")
target_link_libraries(halo2_sky_stencil PUBLIC halo2_glm)
add_executable(vulkan_portability_tests "${HALO2_ROOT}/tests/vulkan_portability_tests.cpp")
target_include_directories(vulkan_portability_tests PRIVATE "${HALO2_ROOT}/include" "${HALO2_ROOT}/src")
target_link_libraries(vulkan_portability_tests PRIVATE Vulkan::Vulkan)
add_test(NAME vulkan_portability COMMAND vulkan_portability_tests)
add_executable(macos_sky_stencil_tests sky_stencil_tests.cpp)
target_link_libraries(macos_sky_stencil_tests PRIVATE halo2_sky_stencil nlohmann_json::nlohmann_json)
target_compile_definitions(macos_sky_stencil_tests PRIVATE
    DEPTH_SHADER_DIR="${CMAKE_BINARY_DIR}/shaders" SKY_REFERENCE_ROOT="${HALO2_SKY_REFERENCE_ROOT}")
if(APPLE)
    foreach(arch IN LISTS CMAKE_OSX_ARCHITECTURES)
        add_test(NAME macos_sky_stencil_${arch} COMMAND /usr/bin/arch -${arch}
            $<TARGET_FILE:macos_sky_stencil_tests> "${CMAKE_BINARY_DIR}/sky-stencil-${arch}.json")
        set_tests_properties(macos_sky_stencil_${arch} PROPERTIES RUN_SERIAL TRUE TIMEOUT 180)
    endforeach()
else()
    add_test(NAME macos_sky_stencil_host COMMAND macos_sky_stencil_tests "${CMAKE_BINARY_DIR}/sky-stencil-host.json")
    set_tests_properties(macos_sky_stencil_host PROPERTIES RUN_SERIAL TRUE TIMEOUT 180)
endif()
file(COPY "${HALO2_SKY_REFERENCE_ROOT}/reference-receipt.json" DESTINATION "${CMAKE_BINARY_DIR}")

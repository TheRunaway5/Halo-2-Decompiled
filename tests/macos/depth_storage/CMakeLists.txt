cmake_minimum_required(VERSION 3.24)
get_filename_component(HALO2_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${HALO2_ROOT}/cmake/macos/Defaults.cmake")
project(Halo2DepthStorageTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(nlohmann_json REQUIRED CONFIG)
include("${HALO2_ROOT}/cmake/Glm.cmake")
include("${HALO2_ROOT}/cmake/TestStorage.cmake")
include(CTest)
if(APPLE)
    include("${HALO2_ROOT}/cmake/macos/Platform.cmake")
else()
    find_package(Vulkan REQUIRED)
endif()
find_program(GLSLANG_VALIDATOR glslangValidator REQUIRED)
include("${HALO2_ROOT}/cmake/SourceWeatherDepthAlias.cmake")
halo2_add_source_weather_depth_alias("${HALO2_ROOT}")
target_link_libraries(halo2_weather_depth_alias PUBLIC halo2_glm)
add_executable(vulkan_portability_tests "${HALO2_ROOT}/tests/vulkan_portability_tests.cpp")
target_include_directories(vulkan_portability_tests PRIVATE "${HALO2_ROOT}/include" "${HALO2_ROOT}/src")
target_link_libraries(vulkan_portability_tests PRIVATE Vulkan::Vulkan)
add_test(NAME vulkan_portability COMMAND vulkan_portability_tests)
add_executable(depth_storage_tests depth_storage_tests.cpp)
target_link_libraries(depth_storage_tests PRIVATE halo2_weather_depth_alias nlohmann_json::nlohmann_json)
target_compile_definitions(depth_storage_tests PRIVATE DEPTH_SHADER_DIR="${CMAKE_BINARY_DIR}/shaders")
set(probe_shaders)
foreach(stage vert frag)
    set(shader "${CMAKE_BINARY_DIR}/shaders/depth_color.${stage}.spv")
    add_custom_command(OUTPUT "${shader}"
        COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/shaders"
        COMMAND "${GLSLANG_VALIDATOR}" -V --target-env vulkan1.1
            "${CMAKE_CURRENT_SOURCE_DIR}/depth_color.${stage}" -o "${shader}"
        DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/depth_color.${stage}" VERBATIM)
    list(APPEND probe_shaders "${shader}")
endforeach()
add_custom_target(depth_storage_probe_shaders DEPENDS ${probe_shaders})
add_dependencies(depth_storage_tests depth_storage_probe_shaders)
if(APPLE)
    foreach(arch IN LISTS CMAKE_OSX_ARCHITECTURES)
        add_test(NAME depth_storage_${arch} COMMAND /usr/bin/arch -${arch}
            $<TARGET_FILE:depth_storage_tests> "${CMAKE_BINARY_DIR}/depth-storage-${arch}.json")
        set_tests_properties(depth_storage_${arch} PROPERTIES RUN_SERIAL TRUE TIMEOUT 240)
    endforeach()
else()
    add_test(NAME depth_storage_host COMMAND depth_storage_tests "${CMAKE_BINARY_DIR}/depth-storage-host.json")
    set_tests_properties(depth_storage_host PROPERTIES RUN_SERIAL TRUE TIMEOUT 240)
endif()

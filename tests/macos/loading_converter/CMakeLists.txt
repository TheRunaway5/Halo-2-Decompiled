cmake_minimum_required(VERSION 3.24)
get_filename_component(HALO2_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
if(APPLE)
    include("${HALO2_ROOT}/cmake/macos/Defaults.cmake")
endif()
project(Halo2LoadingConverterMetadata LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(glm CONFIG REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
include(CTest)
include("${HALO2_ROOT}/cmake/TestStorage.cmake")
add_executable(loading_converter_metadata_tests metadata_tests.cpp
    "${HALO2_ROOT}/src/audio/mcc_loading_converter_metadata.cpp")
target_include_directories(loading_converter_metadata_tests PRIVATE "${HALO2_ROOT}/include" "${HALO2_ROOT}/src")
target_link_libraries(loading_converter_metadata_tests PRIVATE nlohmann_json::nlohmann_json glm::glm-header-only)
if(APPLE)
    foreach(arch IN LISTS CMAKE_OSX_ARCHITECTURES)
        add_test(NAME loading_converter_metadata_${arch}
            COMMAND /usr/bin/arch -${arch} $<TARGET_FILE:loading_converter_metadata_tests>)
    endforeach()
else()
    add_test(NAME loading_converter_metadata COMMAND loading_converter_metadata_tests)
endif()
add_test(NAME loading_converter_python_policy COMMAND ${Python3_EXECUTABLE} -B
    "${HALO2_ROOT}/tests/macos/test_loading_converter.py")

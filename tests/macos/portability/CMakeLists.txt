cmake_minimum_required(VERSION 3.24)
get_filename_component(HALO2_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${HALO2_ROOT}/cmake/macos/Defaults.cmake")
project(Halo2MacOSSharedPortabilityTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(SDL2 REQUIRED CONFIG)
include("${HALO2_ROOT}/cmake/Glm.cmake")
find_package(nlohmann_json REQUIRED CONFIG)
find_package(Threads REQUIRED)
include("${HALO2_ROOT}/cmake/TestStorage.cmake")
include(CTest)
add_library(macos_shared_material_objects OBJECT
    "${HALO2_ROOT}/src/assets/appearance.cpp" "${HALO2_ROOT}/src/ui/frontend_model.cpp"
    "${HALO2_ROOT}/src/gameplay/objects/object_lighting.cpp" "${HALO2_ROOT}/src/rendering/world/world_material.cpp"
    "${HALO2_ROOT}/src/gameplay/units/player_armor_emblem.cpp")
target_include_directories(macos_shared_material_objects PRIVATE "${HALO2_ROOT}/include" "${HALO2_ROOT}/src"
    $<TARGET_PROPERTY:SDL2::SDL2,INTERFACE_INCLUDE_DIRECTORIES>)
target_link_libraries(macos_shared_material_objects PRIVATE halo2_glm nlohmann_json::nlohmann_json)
set(targets texture_residency_tests material_pass_residency_tests renderer_image_residency_tests)
foreach(target IN LISTS targets)
    add_executable(${target} "${HALO2_ROOT}/tests/${target}.cpp")
    target_include_directories(${target} PRIVATE "${HALO2_ROOT}/include" "${HALO2_ROOT}/src")
    target_link_libraries(${target} PRIVATE halo2_glm nlohmann_json::nlohmann_json Threads::Threads)
    if(APPLE)
        foreach(arch IN LISTS CMAKE_OSX_ARCHITECTURES)
            add_test(NAME ${target}_${arch} COMMAND /usr/bin/arch -${arch} $<TARGET_FILE:${target}>)
        endforeach()
    else()
        add_test(NAME ${target} COMMAND ${target})
    endif()
endforeach()
add_custom_target(macos_shared_portability_tests DEPENDS ${targets} macos_shared_material_objects)

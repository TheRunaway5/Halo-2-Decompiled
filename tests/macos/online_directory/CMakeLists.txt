cmake_minimum_required(VERSION 3.24)
get_filename_component(HALO2_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${HALO2_ROOT}/cmake/macos/Defaults.cmake")
project(Halo2MacOSOnlineDirectoryTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
find_package(CURL 7.86 REQUIRED)
find_package(nlohmann_json 3.10 REQUIRED)
include("${HALO2_ROOT}/cmake/TestStorage.cmake")
include(CTest)
add_executable(online_directory_tests "${HALO2_ROOT}/tests/online_directory_tests.cpp"
    "${HALO2_ROOT}/src/network/online_directory.cpp")
target_include_directories(online_directory_tests PRIVATE "${HALO2_ROOT}/include" "${HALO2_ROOT}/src")
target_link_libraries(online_directory_tests PRIVATE CURL::libcurl nlohmann_json::nlohmann_json Threads::Threads)
if(APPLE)
    add_custom_command(TARGET online_directory_tests POST_BUILD
        COMMAND /usr/bin/lipo -archs $<TARGET_FILE:online_directory_tests>)
    foreach(arch IN LISTS CMAKE_OSX_ARCHITECTURES)
        add_test(NAME online_directory_client_${arch} COMMAND /usr/bin/arch -${arch}
            $<TARGET_FILE:online_directory_tests>)
        set_tests_properties(online_directory_client_${arch} PROPERTIES TIMEOUT 45)
    endforeach()
else()
    add_test(NAME online_directory_client_host COMMAND online_directory_tests)
    set_tests_properties(online_directory_client_host PROPERTIES TIMEOUT 45)
endif()

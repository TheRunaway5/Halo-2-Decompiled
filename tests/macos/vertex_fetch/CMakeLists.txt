cmake_minimum_required(VERSION 3.24)
get_filename_component(HALO2_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${HALO2_ROOT}/cmake/macos/Defaults.cmake")
project(Halo2MacOSVertexFetchTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(nlohmann_json REQUIRED CONFIG)
include("${HALO2_ROOT}/cmake/TestStorage.cmake")
include(CTest)
if(APPLE)
    include("${HALO2_ROOT}/cmake/macos/Platform.cmake")
else()
    find_package(Vulkan REQUIRED)
endif()
find_program(GLSLANG_VALIDATOR glslangValidator REQUIRED)
find_program(SPIRV_VALIDATOR spirv-val REQUIRED)
set(shaders)
foreach(name fetch.vert fetch_integer.vert fetch.frag)
    set(output "${CMAKE_BINARY_DIR}/shaders/${name}.spv")
    set(defines)
    if(name STREQUAL "fetch_integer.vert")
        set(source fetch.vert)
        set(defines -DH2_INTEGER_INPUT=1)
    else()
        set(source ${name})
    endif()
    add_custom_command(OUTPUT "${output}"
        COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/shaders"
        COMMAND "${GLSLANG_VALIDATOR}" -V --target-env vulkan1.1 "-I${HALO2_ROOT}/shaders" ${defines}
            "${CMAKE_CURRENT_SOURCE_DIR}/${source}" -o "${output}"
        COMMAND "${SPIRV_VALIDATOR}" --target-env vulkan1.1 "${output}"
        DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${source}" "${HALO2_ROOT}/shaders/source_world_vertex_fetch.glsl" VERBATIM)
    list(APPEND shaders "${output}")
endforeach()
add_custom_target(vertex_fetch_shaders DEPENDS ${shaders})
add_executable(vertex_fetch_tests vertex_fetch_tests.cpp)
target_link_libraries(vertex_fetch_tests PRIVATE Vulkan::Vulkan nlohmann_json::nlohmann_json)
target_include_directories(vertex_fetch_tests PRIVATE "${HALO2_ROOT}/include" "${HALO2_ROOT}/src")
target_compile_definitions(vertex_fetch_tests PRIVATE DEPTH_SHADER_DIR="${CMAKE_BINARY_DIR}/shaders")
add_dependencies(vertex_fetch_tests vertex_fetch_shaders)
add_executable(vulkan_portability_tests "${HALO2_ROOT}/tests/vulkan_portability_tests.cpp")
target_include_directories(vulkan_portability_tests PRIVATE "${HALO2_ROOT}/include" "${HALO2_ROOT}/src")
target_link_libraries(vulkan_portability_tests PRIVATE Vulkan::Vulkan)
add_test(NAME vulkan_portability COMMAND vulkan_portability_tests)
if(APPLE)
    foreach(arch IN LISTS CMAKE_OSX_ARCHITECTURES)
        add_test(NAME macos_vertex_fetch_${arch} COMMAND /usr/bin/arch -${arch}
            $<TARGET_FILE:vertex_fetch_tests> "${CMAKE_BINARY_DIR}/vertex-fetch-${arch}.json")
        set_tests_properties(macos_vertex_fetch_${arch} PROPERTIES RUN_SERIAL TRUE TIMEOUT 180)
    endforeach()
else()
    add_test(NAME macos_vertex_fetch_host COMMAND vertex_fetch_tests "${CMAKE_BINARY_DIR}/vertex-fetch-host.json")
    set_tests_properties(macos_vertex_fetch_host PROPERTIES RUN_SERIAL TRUE TIMEOUT 180)
endif()
